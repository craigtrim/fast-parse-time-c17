cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/fast_parse_time.h" FPT_VERSION_DEFINE
    REGEX "^#define FPT_VERSION ")
string(REGEX MATCH "\"([0-9]+)\\.([0-9]+)\\.([0-9]+)-c17\\.([0-9]+)\""
    FPT_VERSION_MATCH "${FPT_VERSION_DEFINE}")
if(NOT FPT_VERSION_MATCH)
    message(FATAL_ERROR "FPT_VERSION must use MAJOR.MINOR.PATCH-c17.REVISION")
endif()
set(FPT_SOURCE_VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
set(FPT_FULL_VERSION "${FPT_SOURCE_VERSION}-c17.${CMAKE_MATCH_4}")
set(FPT_WINDOWS_VERSION "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},${CMAKE_MATCH_4}")
project(fast_parse_time VERSION ${FPT_SOURCE_VERSION} LANGUAGES C)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS include/fast_parse_time.h)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(FPT_BUILD_SHARED "Build the C shared library" ON)
option(FPT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(FPT_SANITIZE AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_UNICODE ON CACHE BOOL "" FORCE)
set(PCRE2_STATIC_PIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/pcre2-10.45 EXCLUDE_FROM_ALL)
set_target_properties(pcre2-8-static PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/vendor/pcre2-10.45>"
    EXPORT_NAME pcre2-8)
set(FPT_SOURCES src/core.c src/engine.c src/explicit.c src/relative.c src/datetime.c src/calendar.c)
add_library(fast_parse_time STATIC ${FPT_SOURCES})
add_library(fast_parse_time::fast_parse_time ALIAS fast_parse_time)
if(FPT_BUILD_SHARED)
    add_library(fast_parse_time_shared SHARED ${FPT_SOURCES})
    set_target_properties(fast_parse_time_shared PROPERTIES OUTPUT_NAME fast_parse_time)
    target_compile_definitions(fast_parse_time_shared PRIVATE FPT_BUILD PUBLIC FPT_SHARED)
endif()
foreach(target fast_parse_time fast_parse_time_shared)
    if(TARGET ${target})
        target_include_directories(${target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/vendor/pcre2-10.45 ${CMAKE_CURRENT_SOURCE_DIR}/vendor/pcre2-10.45/src)
        target_link_libraries(${target} PRIVATE pcre2-8-static)
        if(NOT WIN32)
            target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L _GNU_SOURCE)
            target_link_libraries(${target} PRIVATE m)
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wno-overlength-strings)
            if(FPT_SANITIZE)
                target_link_options(${target} PUBLIC -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()
add_executable(fpt src/cli.c)
target_link_libraries(fpt PRIVATE fast_parse_time)
if(WIN32)
    enable_language(RC)
    configure_file(cmake/fpt_version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/fpt_version.rc @ONLY)
    target_sources(fpt PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/fpt_version.rc)
    target_compile_definitions(fpt PRIVATE $<$<CONFIG:Debug>:_DEBUG>)
    target_link_libraries(fpt PRIVATE shell32)
endif()
if(MINGW)
    target_link_options(fpt PRIVATE -static)
    if(TARGET fast_parse_time_shared)
        target_link_options(fast_parse_time_shared PRIVATE -static-libgcc)
    endif()
endif()
install(TARGETS fast_parse_time pcre2-8-static fpt EXPORT fast_parse_timeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(TARGET fast_parse_time_shared)
    install(TARGETS fast_parse_time_shared EXPORT fast_parse_timeTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(FILES include/fast_parse_time.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE THIRD_PARTY_NOTICES.md vendor/pcre2-10.45/LICENCE.md vendor/PYTHON-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/fast_parse_time)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/fast_parse_timeConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/fast_parse_timeConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fast_parse_time)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/fast_parse_timeConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT fast_parse_timeTargets NAMESPACE fast_parse_time:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fast_parse_time)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fast_parse_timeConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/fast_parse_timeConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fast_parse_time)
if(BUILD_TESTING)
    add_executable(fpt_tests tests/test_native.c)
    target_link_libraries(fpt_tests PRIVATE fast_parse_time)
    add_test(NAME native COMMAND fpt_tests)
    find_package(Threads REQUIRED)
    add_executable(fpt_thread_tests tests/test_threads.c)
    target_link_libraries(fpt_thread_tests PRIVATE fast_parse_time Threads::Threads)
    add_test(NAME shared_context COMMAND fpt_thread_tests)
endif()
